    | needs input
      Worker --> Agent: Ask for context
  gap Next step

Embed the same source two ways.

Use the custom element for a page, or call the renderer directly when your application owns the container. 

import { defineLinesAndArrows } from "@lines-and-arrows/core";

defineLinesAndArrows();

const diagram = document.querySelector("lines-and-arrows");
diagram.source = `
@Agent
@Tool

Agent -> Tool: Run
Tool --> Agent: Done
`;

<lines-and-arrows
  mode="view"
  theme="auto"
></lines-and-arrows>
import {
  parse,
  renderDiagram,
} from "@lines-and-arrows/core";

const diagramDocument = parse(source);
const view = renderDiagram(container, diagramDocument, {
  theme: "auto",
});

view.select("actor:Tool");
view.destroy();
